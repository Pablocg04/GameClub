cmake_minimum_required(VERSION 3.22)
project(gameclub_native LANGUAGES CXX)
set(CMAKE_CXX_STANDARD 17)
if(ANDROID)
  find_package(SDL3 REQUIRED CONFIG)
  add_library(gameclub_native SHARED player.cpp)
  add_library(gameclub_pinball SHARED pinball_android.cpp)
  target_link_libraries(gameclub_pinball PRIVATE SDL3::SDL3 dl log)
else()
  find_package(SDL3 REQUIRED CONFIG PATHS "${CMAKE_CURRENT_SOURCE_DIR}/vendor/windows/SDL3-3.4.18/cmake" NO_DEFAULT_PATH)
  add_executable(gameclub_native player.cpp)
  target_sources(gameclub_native PRIVATE pinball_host.cpp)
  target_link_libraries(gameclub_native PRIVATE user32)
  if(TARGET gameclub)
    set(NATIVE_INSTALL_DIR "$<TARGET_FILE_DIR:gameclub>")
  else()
    set(NATIVE_INSTALL_DIR ".")
  endif()
  install(TARGETS gameclub_native RUNTIME DESTINATION "${NATIVE_INSTALL_DIR}" COMPONENT Runtime)
  install(FILES $<TARGET_FILE:SDL3::SDL3-shared> DESTINATION "${NATIVE_INSTALL_DIR}" COMPONENT Runtime)
endif()
target_include_directories(gameclub_native PRIVATE "${CMAKE_CURRENT_SOURCE_DIR}/vendor")
target_link_libraries(gameclub_native PRIVATE SDL3::SDL3)
option(GAMECLUB_LAYOUT_TESTS "Comprobar diseños táctiles móviles" OFF)
if(GAMECLUB_LAYOUT_TESTS)
  enable_testing()
  add_executable(gameclub_touch_layout_test touch_layout_test.cpp)
  add_test(NAME touch_layout COMMAND gameclub_touch_layout_test)
  add_executable(gameclub_pad_input_test pad_input_test.cpp)
  add_test(NAME pad_input COMMAND gameclub_pad_input_test)
endif()
