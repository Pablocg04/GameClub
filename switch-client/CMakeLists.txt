cmake_minimum_required(VERSION 3.22)
project(gameclub_switch_preview LANGUAGES CXX)
set(CMAKE_CXX_STANDARD 17)
file(READ "${CMAKE_SOURCE_DIR}/../client/pubspec.yaml" GAMECLUB_PUBSPEC)
set_property(DIRECTORY APPEND PROPERTY CMAKE_CONFIGURE_DEPENDS "${CMAKE_SOURCE_DIR}/../client/pubspec.yaml")
string(REGEX MATCH "version: ([0-9]+\\.[0-9]+\\.[0-9]+)" _ "${GAMECLUB_PUBSPEC}")
add_compile_definitions(GAMECLUB_SWITCH_VERSION="${CMAKE_MATCH_1}")
find_package(SDL2 REQUIRED CONFIG)
find_package(SDL2_image REQUIRED CONFIG)
add_executable(gameclub_switch_preview src/main.cpp src/http.cpp src/pinball.cpp vendor/miniz/miniz.c)
set_source_files_properties(vendor/miniz/miniz.c PROPERTIES LANGUAGE CXX)
target_include_directories(gameclub_switch_preview PRIVATE vendor)
target_compile_options(gameclub_switch_preview PRIVATE /utf-8 /W4 /EHsc)
target_compile_definitions(gameclub_switch_preview PRIVATE _CRT_SECURE_NO_WARNINGS MINIZ_NO_ZLIB_APIS)
target_link_libraries(gameclub_switch_preview PRIVATE SDL2::SDL2 SDL2_image::SDL2_image winhttp)
add_custom_command(TARGET gameclub_switch_preview POST_BUILD
    COMMAND ${CMAKE_COMMAND} -E copy_if_different $<TARGET_FILE:SDL2::SDL2> $<TARGET_FILE_DIR:gameclub_switch_preview>
    COMMAND ${CMAKE_COMMAND} -E copy_if_different $<TARGET_FILE:SDL2_image::SDL2_image> $<TARGET_FILE_DIR:gameclub_switch_preview>)
add_executable(gameclub_switch_test tests/model_test.cpp src/http.cpp src/pinball.cpp vendor/miniz/miniz.c)
target_include_directories(gameclub_switch_test PRIVATE vendor src)
target_compile_features(gameclub_switch_test PRIVATE cxx_std_20)
target_compile_options(gameclub_switch_test PRIVATE /utf-8 /W4 /EHsc)
target_compile_definitions(gameclub_switch_test PRIVATE _CRT_SECURE_NO_WARNINGS MINIZ_NO_ZLIB_APIS)
target_link_libraries(gameclub_switch_test PRIVATE winhttp)
enable_testing()
add_test(NAME switch_model COMMAND gameclub_switch_test)
if(EXISTS "${CMAKE_SOURCE_DIR}/../dist/switch-ports/vpinball/third-party/include/ThreadPool.h")
  add_executable(gameclub_switch_threads_test ../switch-pinball/tests/thread_pool_test.cpp)
  target_include_directories(gameclub_switch_threads_test PRIVATE ../dist/switch-ports/vpinball/third-party/include)
  target_compile_definitions(gameclub_switch_threads_test PRIVATE __SWITCH__)
  target_compile_options(gameclub_switch_threads_test PRIVATE /utf-8 /EHsc)
  add_test(NAME switch_threads COMMAND gameclub_switch_threads_test)
  set_tests_properties(switch_threads PROPERTIES TIMEOUT 30)
endif()
if(EXISTS "${CMAKE_SOURCE_DIR}/../dist/switch-ports/prefix/include/SDL3/SDL_pixels.h")
  add_executable(gameclub_switch_format_test ../switch-pinball/tests/display_format_test.cpp)
  target_include_directories(gameclub_switch_format_test PRIVATE ../switch-pinball/include ../dist/switch-ports/prefix/include)
  target_compile_options(gameclub_switch_format_test PRIVATE /utf-8 /EHsc)
  add_test(NAME switch_display_format COMMAND gameclub_switch_format_test)
endif()
